# correctionlib from the LCG view (or any installation on the CMAKE_PREFIX_PATH)
find_path(CORRECTIONLIB_INCLUDE_DIR correction.h)
find_library(CORRECTIONLIB_LIBRARY correctionlib)
if(NOT CORRECTIONLIB_INCLUDE_DIR OR NOT CORRECTIONLIB_LIBRARY)
  execute_process(COMMAND python3 -m correctionlib.config --incdir OUTPUT_VARIABLE _cl_inc OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  execute_process(COMMAND python3 -m correctionlib.config --libdir OUTPUT_VARIABLE _cl_lib OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  find_path(CORRECTIONLIB_INCLUDE_DIR correction.h HINTS ${_cl_inc})
  find_library(CORRECTIONLIB_LIBRARY correctionlib HINTS ${_cl_lib})
endif()
if(NOT CORRECTIONLIB_INCLUDE_DIR OR NOT CORRECTIONLIB_LIBRARY)
  message(FATAL_ERROR "[HELL] correctionlib not found (correction.h / libcorrectionlib): use the LCG environment")
endif()
message(STATUS "[HELL] correctionlib: ${CORRECTIONLIB_LIBRARY}")
hell_add_library(Corrections SOURCES src/CorrectionService.cc DEPENDS HellCore ${CORRECTIONLIB_LIBRARY})
target_include_directories(HellCorrections PUBLIC ${CORRECTIONLIB_INCLUDE_DIR})
